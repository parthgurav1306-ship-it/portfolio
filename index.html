#include <iostream>
#include <vector>

void quickSort(std::vector<int>& arr, int low, int high) {
  if (low >= high) return;
  int pivot = arr[high], i = low - 1;
  for (int j = low; j < high; ++j) {
    if (arr[j] <= pivot) std::swap(arr[++i], arr[j]);
  }
  std::swap(arr[i + 1], arr[high]);
  quickSort(arr, low, i);
  quickSort(arr, i + 2, high);
}
